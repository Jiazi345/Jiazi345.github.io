kernel是什么？

在 computeshader中：

[numthreads(8,8,1)] 代表8行8列的一组像素

在C#中 shader.Dispatch(int kernelid, int x, int y, int n);代表绘制x行y列n层的kernel

shader.FindKernel(string KernalName); 从名字获取kernel的id

如：在256*256的render texture中， 绘制[numthreads(8,8,1)]的kernel,x=32 y=32 n=1 则铺满整个纹理 

#pragma kernel SoildColor

RWTexture2D<float4> Result;

[numthreads(8,8,1)]
void SoildColor(uint3 id: SV_DISPATCHTHREADID)
{
    Result[id.xy]=float4(1,1,0,1);
}
public ComputeShader shader;
    public int Resolution = 256;
    public string KernalName;
    Renderer rend;
    RenderTexture outputTexture;
    int kernelHandle;

 void Start()
    {
        outputTexture = new RenderTexture(Resolution, Resolution, 0);
        outputTexture.enableRandomWrite=true;
        outputTexture.Create();

        rend = GetComponent<Renderer>();
        rend.enabled = true;
        InitShader();
    }

    private void InitShader()
    {
        kernelHandle = shader.FindKernel(KernalName);

        shader.SetTexture(kernelHandle, "Result", outputTexture);
        rend.material.SetTexture("_BaseMap", outputTexture);
        DispatchShader(Resolution / 8, Resolution / 8);
    }

    private void DispatchShader(int x, int y)
    {
        shader.Dispatch(kernelHandle, x, y, 1);
    }
